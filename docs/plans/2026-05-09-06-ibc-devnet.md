```go
package e2e_test

import (
	"context"
	"testing"

	"github.com/strangelove-ventures/interchaintest/v8"
	"github.com/strangelove-ventures/interchaintest/v8/chain/cosmos"
	"github.com/strangelove-ventures/interchaintest/v8/ibc"
	"github.com/strangelove-ventures/interchaintest/v8/testreporter"
	"github.com/stretchr/testify/require"
	"go.uber.org/zap/zaptest"
)

// TestIBCTransferVTX verifies that VTX (uvtx) can be transferred cross-chain via ICS-20.
func TestIBCTransferVTX(t *testing.T) {
	if testing.Short() {
		t.Skip("skipping IBC E2E test in short mode")
	}

	ctx := context.Background()
	logger := zaptest.NewLogger(t)
	reporter := testreporter.NewNopReporter()
	eRep := reporter.RelayerExecReporter(t)

	chainConfig := ibc.ChainConfig{
		Type:           "cosmos",
		Name:           "vertix",
		ChainID:        "vertix-e2e-1",
		Images:         []ibc.DockerImage{{Repository: "vertixd", Version: "local", UIDGid: "1025:1025"}},
		Bin:            "vertixd",
		Bech32Prefix:   "vtx",
		Denom:          "uvtx",
		GasPrices:      "0.025uvtx",
		GasAdjustment:  1.5,
		TrustingPeriod: "336h",
		NoHostMount:    false,
	}

	// Two Vertix chains for cross-chain transfer test
	cf := interchaintest.NewBuiltinChainFactory(logger, []*interchaintest.ChainSpec{
		{Name: "vertix-a", ChainConfig: chainConfig, NumValidators: interchaintest.IntPtr(1)},
		{Name: "vertix-b", ChainConfig: chainConfig, NumValidators: interchaintest.IntPtr(1)},
	})

	chains, err := cf.Chains(t.Name())
	require.NoError(t, err)
	chainA, chainB := chains[0].(*cosmos.CosmosChain), chains[1].(*cosmos.CosmosChain)

	// Use Hermes relayer
	relayer := interchaintest.NewBuiltinRelayerFactory(ibc.CosmosRly, logger).Build(
		t, nil, testreporter.NewNopReporter(),
	)

	ic := interchaintest.NewInterchain().
		AddChain(chainA).
		AddChain(chainB).
		AddRelayer(relayer, "hermes").
		AddLink(interchaintest.InterchainLink{
			Chain1:  chainA,
			Chain2:  chainB,
			Relayer: relayer,
			Path:    "vertix-a-to-b",
		})

	require.NoError(t, ic.Build(ctx, eRep, interchaintest.InterchainBuildOptions{
		TestName:  t.Name(),
		Client:    nil,
		NetworkID: "vertix-e2e",
	}))
	defer ic.Close()

	// Fund a user on chain A
	users := interchaintest.GetAndFundTestUsers(t, ctx, "user", 1_000_000_000, chainA, chainB)
	userA, userB := users[0], users[1]

	// Get the IBC transfer channel
	channelA, err := relayer.GetChannels(ctx, eRep, chainA.Config().ChainID)
	require.NoError(t, err)
	require.NotEmpty(t, channelA)

	// ICS-20 transfer: 100 VTX from chain A to chain B
	transferAmount := ibc.WalletAmount{
		Address: userB.FormattedAddress(),
		Denom:   "uvtx",
		Amount:  100_000_000, // 100 VTX in uvtx
	}
	tx, err := chainA.SendIBCTransfer(ctx, channelA[0].ChannelID, userA.KeyName(), transferAmount, ibc.TransferOptions{})
	require.NoError(t, err)
	require.NotEmpty(t, tx.TxHash)

	// Wait for relayer to relay the packet
	require.NoError(t, relayer.StartRelayer(ctx, eRep, "vertix-a-to-b"))
	defer relayer.StopRelayer(ctx, eRep)

	// Verify chain B received the IBC denom
	ibcDenom := transfertypes.GetDenomPrefix(channelA[0].Counterparty.PortID, channelA[0].Counterparty.ChannelID) + "uvtx"
	balB, err := chainB.GetBalance(ctx, userB.FormattedAddress(), ibcDenom)
	require.NoError(t, err)
	require.Equal(t, int64(100_000_000), balB.Int64(), "IBC transferred VTX must appear on chain B")
}
```

# Vertix IBC Relayer Setup

## Hermes (recommended)

### Install Hermes v1.8.x

```bash
# Download release binary
curl -L "https://github.com/informalsystems/hermes/releases/download/v1.8.0/hermes-v1.8.0-x86_64-unknown-linux-gnu.tar.gz" | tar xz

hermes version
```

### Configure

```bash
mkdir -p ~/.hermes
cp infra/hermes/config.toml ~/.hermes/config.toml
# Edit ~/.hermes/config.toml to update your node endpoints and RPC addresses
```

### Add Relayer Keys

```bash
# Add Vertix relayer key (fund this address with uvtx for gas)
hermes keys add --chain vertix-1 --mnemonic-file /path/to/relayer.mnemonic

# Add Cosmos Hub relayer key
hermes keys add --chain cosmoshub-4 --mnemonic-file /path/to/relayer.mnemonic
```

### Validate Configuration

```bash
hermes config validate
hermes health-check
```

### Create IBC Channel (Vertix ↔ Cosmos Hub)

```bash
# Create client, connection, and channel in one command
hermes create channel \
  --a-chain vertix-1 \
  --b-chain cosmoshub-4 \
  --a-port transfer \
  --b-port transfer \
  --new-connection
```

Expected output: `SUCCESS Channel { a_side: ... channel_id: "channel-0" }`

### Start Relayer

```bash
hermes start
```

### Run as Systemd Service

```ini
[Unit]

After=network.target

[Service]
User=relayer
ExecStart=/usr/local/bin/hermes start
Restart=on-failure
RestartSec=5s
Environment=RUST_LOG=info

[Install]
WantedBy=multi-user.target
```

```

## Client Refresh Strategy

IBC clients expire if not updated. Configure Hermes client refresh:

```toml
[mode.clients]
enabled = true
refresh = true    # automatically update expiring clients
```

Clients expire after `trusting_period` (336h = 14 days for Vertix). The relayer should be online continuously, or clients will expire and channels will need to be recreated.

## Monitoring

Hermes exposes metrics at `http://localhost:3001/metrics`. Key metrics to alert on:

| Metric | Alert threshold |
|---|---|
| `hermes_client_expired_total` | > 0 |
| `hermes_packet_send_errors_total` | > 5 per hour |
| `hermes_wallet_balance` | < 10 VTX |
```

Create `scripts/devnet-init.sh`:

```bash
#!/usr/bin/env bash
set -euo pipefail

BINARY="vertixd"
CHAIN_ID="vertix-devnet-1"
DENOM="uvtx"
KEYRING="test"

# Initialize 3 validator nodes
for i in 1 2 3; do
  $BINARY init "validator${i}" --chain-id "$CHAIN_ID" --home "infra/devnet/validator${i}"
done

# Create validator keys
for i in 1 2 3; do
  $BINARY keys add "validator${i}" --keyring-backend "$KEYRING" --home "infra/devnet/validator${i}"
done

# Add genesis accounts to validator1 (genesis coordinator)
for i in 1 2 3; do
  ADDR=$($BINARY keys show "validator${i}" -a --keyring-backend "$KEYRING" --home "infra/devnet/validator${i}")
  $BINARY genesis add-genesis-account "$ADDR" "10000000000000${DENOM}" --home "infra/devnet/validator1"
done

# Add faucet account
$BINARY genesis add-genesis-account vtx1faucetxxxxxxxxxxxxxxxxxxxxxxxxxxxxx "2100000000000${DENOM}" --home "infra/devnet/validator1"

# Collect gentxs
for i in 1 2 3; do
  VALKEY=$($BINARY keys show "validator${i}" --keyring-backend "$KEYRING" --home "infra/devnet/validator${i}" --bech val -a)
  $BINARY genesis gentx "validator${i}" "100000000${DENOM}" \
    --chain-id "$CHAIN_ID" \
    --keyring-backend "$KEYRING" \
    --home "infra/devnet/validator${i}" \
    --output-document "infra/devnet/validator1/config/gentx/gentx-val${i}.json" \
    --commission-rate 0.1 \
    --commission-max-rate 0.2 \
    --commission-max-change-rate 0.01
done

$BINARY genesis collect-gentxs --home "infra/devnet/validator1"
$BINARY genesis validate-genesis --home "infra/devnet/validator1"

# Copy genesis to other validators
for i in 2 3; do
  cp "infra/devnet/validator1/config/genesis.json" "infra/devnet/validator${i}/config/genesis.json"
done

# Get validator1 node ID for peer config
VAL1_NODE_ID=$($BINARY comet show-node-id --home "infra/devnet/validator1")
echo "export VAL1_NODE_ID=${VAL1_NODE_ID}" > infra/devnet/.env

echo "Devnet initialized. Run 'make devnet-up' to start."
```
