# x/fees Module Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement the `x/fees` Cosmos SDK module — an EndBlock hook that collects all accumulated fees from the module fee account, burns 40% permanently via `x/bank.BurnCoins`, and routes 60% to `x/distribution`'s fee pool for distribution to VTX stakers.

**Architecture:** `x/fees` is a thin coordination module with no custom state of its own. At every `EndBlock` it reads the fee collector account balance (`auth.FeeCollectorName`), splits it by the `BurnRatio`/`DistributionRatio` governance params, sends the burn portion to `x/bank.BurnCoins()`, and sends the distribution portion to the distribution fee pool. All ratios are governance-adjustable via `MsgUpdateParams`.

**Tech Stack:** Cosmos SDK v0.50.x, Go 1.22+, `x/bank`, `x/distribution`, `x/auth`

**Prerequisite:** Plan 01 (Chain Foundation) must be complete.

```go
package types

import "cosmossdk.io/math"

var (
	DefaultBurnRatio         = math.LegacyMustNewDecFromStr("0.4")
	DefaultDistributionRatio = math.LegacyMustNewDecFromStr("0.6")
)

func DefaultParams() FeesParams {
	return FeesParams{
		BurnRatio:         DefaultBurnRatio,
		DistributionRatio: DefaultDistributionRatio,
	}
}

func (p FeesParams) Validate() error {
	if p.BurnRatio.IsNegative() || p.BurnRatio.GT(math.LegacyOneDec()) {
		return ErrInvalidBurnRatio.Wrapf("burn_ratio=%s", p.BurnRatio)
	}
	if p.DistributionRatio.IsNegative() || p.DistributionRatio.GT(math.LegacyOneDec()) {
		return ErrInvalidBurnRatio.Wrapf("distribution_ratio=%s", p.DistributionRatio)
	}
	sum := p.BurnRatio.Add(p.DistributionRatio)
	if !sum.Equal(math.LegacyOneDec()) {
		return ErrInvalidBurnRatio.Wrapf("burn_ratio + distribution_ratio must equal 1, got %s", sum)
	}
	return nil
}
```

```go
import (
```

```go
func (k Keeper) GetParams(ctx context.Context) (types.FeesParams, error) {
	s := k.storeService.OpenKVStore(ctx)
	bz, err := s.Get(types.KeyParams)
	if err != nil || bz == nil {
		return types.DefaultParams(), nil
	}
	var p types.FeesParams
	k.cdc.MustUnmarshal(bz, &p)
	return p, nil
}

func (k Keeper) SetParams(ctx context.Context, p types.FeesParams) error {
	if err := p.Validate(); err != nil {
		return err
	}
	s := k.storeService.OpenKVStore(ctx)
	bz := k.cdc.MustMarshal(&p)
	return s.Set(types.KeyParams, bz)
}
```

- [ ] **Step 5: Write split.go**

```go
package keeper

import (
	"context"

	sdk "github.com/cosmos/cosmos-sdk/types"
	authtypes "github.com/cosmos/cosmos-sdk/x/auth/types"
	"github.com/vertix-network/vertix/x/fees/types"
)

// SplitFees reads the fee collector balance, burns BurnRatio, sends DistributionRatio to stakers.
// Called from EndBlock every block.
func (k Keeper) SplitFees(ctx context.Context) error {
	sdkCtx := sdk.UnwrapSDKContext(ctx)
	params, err := k.GetParams(ctx)
	if err != nil {
		return err
	}

	feeAddr := authtypes.NewModuleAddress(k.feeCollectorName)
	fees := k.bankKeeper.GetAllBalances(sdkCtx, feeAddr)
	if fees.IsZero() {
		return nil // nothing to split
	}

	var burnCoins, distCoins sdk.Coins
	for _, coin := range fees {
		burnAmt := params.BurnRatio.MulInt(coin.Amount).TruncateInt()
		distAmt := coin.Amount.Sub(burnAmt) // remainder goes to distribution (avoids rounding loss)

		if burnAmt.IsPositive() {
			burnCoins = append(burnCoins, sdk.NewCoin(coin.Denom, burnAmt))
		}
		if distAmt.IsPositive() {
			distCoins = append(distCoins, sdk.NewCoin(coin.Denom, distAmt))
		}
	}

	// Burn portion
	if len(burnCoins) > 0 {
		if err := k.bankKeeper.BurnCoins(sdkCtx, k.feeCollectorName, burnCoins); err != nil {
			return err
		}
		sdkCtx.EventManager().EmitEvent(sdk.NewEvent(
			types.EventTypeFeeBurned,
			sdk.NewAttribute(types.AttributeKeyBurnedAmount, burnCoins.String()),
		))
	}

	// Distribution portion — send to distribution fee pool via FundCommunityPool
	// (this feeds x/distribution's fee pool which is distributed to stakers each block)
	if len(distCoins) > 0 {
		if err := k.distrKeeper.FundCommunityPool(sdkCtx, distCoins, feeAddr); err != nil {
			return err
		}
		sdkCtx.EventManager().EmitEvent(sdk.NewEvent(
			types.EventTypeFeeDistributed,
			sdk.NewAttribute(types.AttributeKeyDistAmount, distCoins.String()),
		))
	}

	return nil
}
```

```go
type msgServer struct{ Keeper }

func NewMsgServer(k Keeper) types.MsgServer { return msgServer{k} }

func (m msgServer) UpdateParams(ctx context.Context, msg *types.MsgUpdateParams) (*types.MsgUpdateParamsResponse, error) {
	if msg.Authority != m.GetAuthority() {
		return nil, types.ErrUnauthorized.Wrapf("expected %s, got %s", m.GetAuthority(), msg.Authority)

		return nil, err
	}
	return &types.MsgUpdateParamsResponse{}, nil
}
```

```go
import (
```

```go
func TestSplitFeesIntegration(t *testing.T) {
	ctx, k, bk, _ := keepertest.FeesKeeper(t)

	// Mint 1,000,000 uvtx into fee collector
	coins := sdk.NewCoins(sdk.NewInt64Coin("uvtx", 1_000_000))
	err := bk.MintCoins(ctx, authtypes.FeeCollectorName, coins)
	require.NoError(t, err)

	supplyBefore := bk.GetSupply(ctx, "uvtx").Amount

	err = k.SplitFees(ctx)
	require.NoError(t, err)

	supplyAfter := bk.GetSupply(ctx, "uvtx").Amount

	// 40% = 400,000 uvtx should be burned
	burned := supplyBefore.Sub(supplyAfter)

	// Fee collector should now be empty
	remaining := bk.GetAllBalances(ctx, authtypes.NewModuleAddress(authtypes.FeeCollectorName))
	require.True(t, remaining.IsZero(), "fee collector must be empty after split")
}
```
