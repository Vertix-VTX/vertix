# x/rwa Module Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement the `x/rwa` Cosmos SDK module — a protocol-agnostic Real World Asset registry with a four-state lifecycle (DRAFT → ATTESTED → ACTIVE → SETTLED), issuer VTX bonding, oracle price attestation, factory denom minting, transfer restriction enforcement, and fee forwarding to `x/fees`.

**Architecture:** Issuers register asset classes by bonding VTX via `x/bank` escrow. Assets progress through a strict state machine requiring oracle attestation before any tokens are minted. RWA tokens use `x/bank` factory denoms (`rwa/{asset-id}`). Transfer restrictions (allowlist/denylist per asset) are enforced at the module ante/message level. Fees (0.1% notional on mint and settle) are collected and forwarded to the fee collector for `x/fees` to split.

**Tech Stack:** Cosmos SDK v0.50.x, Go 1.22+, `x/bank`, `x/oracle` (keeper interface), `x/fees`

**Prerequisite:** Plans 01, 02, and 03 must be complete. `OracleKeeper.GetPrice()` must be available.

  ASSET_STATUS_DRAFT       = 1;
  ASSET_STATUS_ATTESTED    = 2;
  ASSET_STATUS_ACTIVE      = 3;
  ASSET_STATUS_SETTLED     = 4;

// TransferRestriction defines who can send/receive a RWA denom.
message TransferRestriction {
  bool             allow_all  = 1;  // if true, ignore allowlist

  repeated string  denylist   = 3;  // bech32 addresses blocked

message AssetRecord {
  string          asset_id    = 1;  // unique identifier (UUID or issuer-chosen slug)

package vertix.rwa.v1;

import "cosmos/msg/v1/msg.proto";
import "cosmos_proto/cosmos.proto";
import "cosmos/base/v1beta1/coin.proto";
import "vertix/rwa/v1/types.proto";

option go_package = "github.com/vertix-network/vertix/x/rwa/types";

service Msg {
  option (cosmos.msg.v1.service) = true;

  // RegisterAsset creates a new DRAFT asset and locks the issuer bond.
  rpc RegisterAsset(MsgRegisterAsset) returns (MsgRegisterAssetResponse);

  // AttestAsset links oracle price and transitions DRAFT → ATTESTED.
  rpc AttestAsset(MsgAttestAsset) returns (MsgAttestAssetResponse);

  // MintRWA mints RWA tokens for an ATTESTED asset, transitions to ACTIVE.
  rpc MintRWA(MsgMintRWA) returns (MsgMintRWAResponse);

  // SettleRWA burns RWA tokens and releases the issuer bond, transitions to SETTLED.
  rpc SettleRWA(MsgSettleRWA) returns (MsgSettleRWAResponse);

  // UpdateRestrictions updates transfer restrictions for an ACTIVE asset.
  rpc UpdateRestrictions(MsgUpdateRestrictions) returns (MsgUpdateRestrictionsResponse);

  // UpdateParams is a governance message to update RWA params.
  rpc UpdateParams(MsgUpdateParams) returns (MsgUpdateParamsResponse);
}

message MsgRegisterAsset {
  option (cosmos.msg.v1.signer) = "issuer";
  string issuer      = 1 [(cosmos_proto.scalar) = "cosmos.AddressString"];
  string asset_id    = 2;

}

package types

import (
	"cosmossdk.io/math"
	sdk "github.com/cosmos/cosmos-sdk/types"
)

var (
	DefaultMinIssuerBond  = sdk.NewInt64Coin("uvtx", 10_000_000_000) // 10,000 VTX
	DefaultMintFeeRate    = math.LegacyMustNewDecFromStr("0.001")     // 0.1%
	DefaultSettleFeeRate  = math.LegacyMustNewDecFromStr("0.001")     // 0.1%
)

func DefaultParams() RWAParams {
	return RWAParams{
		MinIssuerBond: DefaultMinIssuerBond,
		MintFeeRate:   DefaultMintFeeRate,
		SettleFeeRate: DefaultSettleFeeRate,
	}
}

func (p RWAParams) Validate() error {
	if p.MinIssuerBond.IsNegative() || p.MinIssuerBond.IsZero() {
		return ErrInsufficientBond.Wrap("min_issuer_bond must be positive")
	}
	if p.MintFeeRate.IsNegative() || p.MintFeeRate.GT(math.LegacyOneDec()) {
		return ErrInvalidAmount.Wrapf("mint_fee_rate must be in [0,1], got %s", p.MintFeeRate)
	}
	if p.SettleFeeRate.IsNegative() || p.SettleFeeRate.GT(math.LegacyOneDec()) {
		return ErrInvalidAmount.Wrapf("settle_fee_rate must be in [0,1], got %s", p.SettleFeeRate)
	}
	return nil
}

package types

import (
	"strings"
	sdk "github.com/cosmos/cosmos-sdk/types"
)

func (m *MsgRegisterAsset) ValidateBasic() error {
	if _, err := sdk.AccAddressFromBech32(m.Issuer); err != nil {
		return ErrNotIssuer.Wrapf("invalid issuer: %s", err)
	}
	if strings.TrimSpace(m.AssetId) == "" {
		return ErrAssetNotFound.Wrap("asset_id must not be empty")
	}
	if len(m.AssetId) > 64 {
		return ErrAssetNotFound.Wrap("asset_id must be <= 64 chars")
	}
	if strings.TrimSpace(m.Name) == "" {
		return ErrAssetNotFound.Wrap("name must not be empty")
	}
	if !strings.Contains(m.OraclePair, ":") {
		return ErrOraclePriceNotFound.Wrapf("oracle_pair must be in format BASE:QUOTE, got %q", m.OraclePair)
	}
	return nil
}

func (m *MsgAttestAsset) ValidateBasic() error {
	if _, err := sdk.AccAddressFromBech32(m.Issuer); err != nil {
		return ErrNotIssuer.Wrapf("invalid issuer: %s", err)
	}
	if strings.TrimSpace(m.AssetId) == "" {
		return ErrAssetNotFound.Wrap("asset_id must not be empty")
	}
	return nil
}

func (m *MsgMintRWA) ValidateBasic() error {
	if _, err := sdk.AccAddressFromBech32(m.Issuer); err != nil {
		return ErrNotIssuer.Wrapf("invalid issuer: %s", err)
	}
	if _, err := sdk.AccAddressFromBech32(m.Recipient); err != nil {
		return ErrNotIssuer.Wrapf("invalid recipient: %s", err)
	}
	if !m.Amount.IsPositive() {
		return ErrInvalidAmount.Wrap("amount must be positive")
	}
	return nil
}

func (m *MsgSettleRWA) ValidateBasic() error {
	if _, err := sdk.AccAddressFromBech32(m.Issuer); err != nil {
		return ErrNotIssuer.Wrapf("invalid issuer: %s", err)
	}
	return nil
}

func (m *MsgUpdateParams) ValidateBasic() error {
	if _, err := sdk.AccAddressFromBech32(m.Authority); err != nil {
		return ErrNotIssuer.Wrapf("invalid authority: %s", err)
	}
	return m.Params.Validate()
}

func (k Keeper) RegisterAsset(ctx context.Context, record types.AssetRecord) error {
	if _, found := k.GetAsset(ctx, record.AssetId); found {
		return types.ErrAssetAlreadyExists.Wrapf("asset_id %q already registered", record.AssetId)
	}
	record.Status = types.ASSET_STATUS_DRAFT
	return k.setAsset(ctx, record)
}

func (k Keeper) AttestAsset(ctx context.Context, assetID, issuer string) error {
	asset, found := k.GetAsset(ctx, assetID)
	if !found {
		return types.ErrAssetNotFound.Wrapf("asset_id: %s", assetID)
	}
	if asset.Issuer != issuer {
		return types.ErrNotIssuer.Wrapf("expected %s, got %s", asset.Issuer, issuer)
	}
	if asset.Status != types.ASSET_STATUS_DRAFT {
		return types.ErrInvalidStatus.Wrapf("can only attest DRAFT assets, got %s", asset.Status)
	}

	price, err := k.oracleKeeper.GetPrice(ctx, asset.OraclePair)
	if err != nil {
		return types.ErrOraclePriceNotFound.Wrapf("pair %s: %s", asset.OraclePair, err)
	}

	sdkCtx := sdk.UnwrapSDKContext(ctx)
	asset.LastPrice = price
	asset.Status = types.ASSET_STATUS_ATTESTED
	asset.UpdatedAt = sdkCtx.BlockTime()

	if err := k.setAsset(ctx, asset); err != nil {
		return err
	}
	sdkCtx.EventManager().EmitEvent(sdk.NewEvent(
		types.EventTypeAssetAttested,
		sdk.NewAttribute(types.AttributeKeyAssetID, assetID),
		sdk.NewAttribute(types.AttributeKeyPrice, price.String()),
	))
	return nil
}

func (k Keeper) SettleAsset(ctx context.Context, assetID, issuer string) error {
	asset, found := k.GetAsset(ctx, assetID)
	if !found {
		return types.ErrAssetNotFound.Wrapf("asset_id: %s", assetID)
	}
	if asset.Issuer != issuer {
		return types.ErrNotIssuer.Wrapf("expected %s, got %s", asset.Issuer, issuer)
	}
	if asset.Status != types.ASSET_STATUS_ACTIVE {
		return types.ErrInvalidStatus.Wrapf("can only settle ACTIVE assets, got %s", asset.Status)
	}
	// Bond release and token burn handled in msg_server.go
	sdkCtx := sdk.UnwrapSDKContext(ctx)
	asset.Status = types.ASSET_STATUS_SETTLED
	asset.UpdatedAt = sdkCtx.BlockTime()
	return k.setAsset(ctx, asset)
}

func (k Keeper) GetAsset(ctx context.Context, assetID string) (types.AssetRecord, bool) {
	s := k.storeService.OpenKVStore(ctx)
	bz, err := s.Get(types.AssetKey(assetID))
	if err != nil || bz == nil {
		return types.AssetRecord{}, false
	}
	var a types.AssetRecord
	k.cdc.MustUnmarshal(bz, &a)
	return a, true
}

func (k Keeper) setAsset(ctx context.Context, record types.AssetRecord) error {
	s := k.storeService.OpenKVStore(ctx)
	bz := k.cdc.MustMarshal(&record)
	if err := s.Set(types.AssetKey(record.AssetId), bz); err != nil {
		return err
	}
	// Update issuer index
	return s.Set(types.ByIssuerKey(record.Issuer, record.AssetId), []byte{1})
}

package keeper

import (
	"context"

	sdk "github.com/cosmos/cosmos-sdk/types"
	"github.com/vertix-network/vertix/x/rwa/types"
)

// IsTransferAllowed checks whether a sender/recipient pair is permitted
// to transact the given RWA denom according to the asset's restrictions.
func (k Keeper) IsTransferAllowed(ctx context.Context, denom, sender, recipient string) bool {
	// Extract asset_id from "rwa/{asset_id}"
	if len(denom) <= 4 || denom[:4] != "rwa/" {
		return true // not a RWA denom — not our concern
	}
	assetID := denom[4:]
	asset, found := k.GetAsset(ctx, assetID)
	if !found {
		return false // unknown asset — deny
	}
	if asset.Status != types.ASSET_STATUS_ACTIVE {
		return false // not active — deny all transfers
	}
	r := asset.Restrictions
	if r.AllowAll {
		// Only check denylist
		for _, denied := range r.Denylist {
			if denied == sender || denied == recipient {
				return false
			}
		}
		return true
	}
	// Check allowlist (sender AND recipient must be on it)
	senderOK, recipientOK := false, false
	for _, allowed := range r.Allowlist {
		if allowed == sender {
			senderOK = true
		}
		if allowed == recipient {
			recipientOK = true
		}
	}
	return senderOK && recipientOK
}

package keeper

import (
	"context"
	"time"

	"cosmossdk.io/math"
	sdk "github.com/cosmos/cosmos-sdk/types"
	authtypes "github.com/cosmos/cosmos-sdk/x/auth/types"
	"github.com/vertix-network/vertix/x/rwa/types"
)

type msgServer struct{ Keeper }

func NewMsgServer(k Keeper) types.MsgServer { return msgServer{k} }

func (m msgServer) RegisterAsset(ctx context.Context, msg *types.MsgRegisterAsset) (*types.MsgRegisterAssetResponse, error) {
	if err := msg.ValidateBasic(); err != nil {
		return nil, err
	}

	params, _ := m.GetParams(ctx)
	issuerAddr, _ := sdk.AccAddressFromBech32(msg.Issuer)
	sdkCtx := sdk.UnwrapSDKContext(ctx)

	// Lock minimum bond from issuer account
	bond := params.MinIssuerBond
	if err := m.bankKeeper.SendCoinsFromAccountToModule(sdkCtx, issuerAddr, types.ModuleName, sdk.NewCoins(bond)); err != nil {
		return nil, types.ErrInsufficientBond.Wrapf("could not lock bond: %s", err)
	}

	record := types.AssetRecord{
		AssetId:      msg.AssetId,
		Issuer:       msg.Issuer,
		Name:         msg.Name,
		Description:  msg.Description,
		OraclePair:   msg.OraclePair,
		Status:       types.ASSET_STATUS_DRAFT,
		Bond:         bond,
		MintedSupply: sdk.NewInt64Coin(types.RWADenom(msg.AssetId), 0),
		LastPrice:    math.LegacyZeroDec(),
		Restrictions: types.TransferRestriction{AllowAll: true},
		CreatedAt:    sdkCtx.BlockTime(),
		UpdatedAt:    sdkCtx.BlockTime(),
	}
	if err := m.RegisterAsset(ctx, record); err != nil {
		return nil, err
	}

	sdkCtx.EventManager().EmitEvent(sdk.NewEvent(
		types.EventTypeAssetRegistered,
		sdk.NewAttribute(types.AttributeKeyAssetID, msg.AssetId),
		sdk.NewAttribute(types.AttributeKeyIssuer, msg.Issuer),
	))
	return &types.MsgRegisterAssetResponse{AssetId: msg.AssetId}, nil
}

func (m msgServer) AttestAsset(ctx context.Context, msg *types.MsgAttestAsset) (*types.MsgAttestAssetResponse, error) {
	if err := msg.ValidateBasic(); err != nil {
		return nil, err
	}
	if err := m.AttestAsset(ctx, msg.AssetId, msg.Issuer); err != nil {
		return nil, err
	}
	asset, _ := m.GetAsset(ctx, msg.AssetId)
	return &types.MsgAttestAssetResponse{PriceAtAttestation: asset.LastPrice.String()}, nil
}

func (m msgServer) MintRWA(ctx context.Context, msg *types.MsgMintRWA) (*types.MsgMintRWAResponse, error) {
	if err := msg.ValidateBasic(); err != nil {
		return nil, err
	}
	asset, found := m.GetAsset(ctx, msg.AssetId)
	if !found {
		return nil, types.ErrAssetNotFound.Wrapf("asset_id: %s", msg.AssetId)
	}
	if asset.Issuer != msg.Issuer {
		return nil, types.ErrNotIssuer.Wrapf("expected %s, got %s", asset.Issuer, msg.Issuer)
	}
	if asset.Status != types.ASSET_STATUS_ATTESTED {
		return nil, types.ErrInvalidStatus.Wrapf("must be ATTESTED to mint, got %s", asset.Status)
	}

	sdkCtx := sdk.UnwrapSDKContext(ctx)
	params, _ := m.GetParams(ctx)

	// Compute fee (0.1% of notional: amount * oracle_price * fee_rate)
	price, _ := m.oracleKeeper.GetPrice(ctx, asset.OraclePair)
	notional := price.MulInt(msg.Amount.Amount)
	feeAmt := params.MintFeeRate.Mul(notional).TruncateInt()
	feeCoins := sdk.NewCoins(sdk.NewCoin("uvtx", feeAmt))

	issuerAddr, _ := sdk.AccAddressFromBech32(msg.Issuer)
	// Collect fee
	if feeAmt.IsPositive() {
		if err := m.bankKeeper.SendCoinsFromAccountToModule(sdkCtx, issuerAddr, authtypes.FeeCollectorName, feeCoins); err != nil {
			return nil, err
		}
	}

	// Mint RWA tokens
	rwaDenom := types.RWADenom(msg.AssetId)
	mintCoins := sdk.NewCoins(sdk.NewCoin(rwaDenom, msg.Amount.Amount))
	if err := m.bankKeeper.MintCoins(sdkCtx, types.ModuleName, mintCoins); err != nil {
		return nil, err
	}

	recipientAddr, _ := sdk.AccAddressFromBech32(msg.Recipient)
	if err := m.bankKeeper.SendCoinsFromModuleToAccount(sdkCtx, types.ModuleName, recipientAddr, mintCoins); err != nil {
		return nil, err
	}

	// Update asset record
	asset.Status = types.ASSET_STATUS_ACTIVE
	asset.MintedSupply.Amount = asset.MintedSupply.Amount.Add(msg.Amount.Amount)
	asset.UpdatedAt = sdkCtx.BlockTime()
	m.setAsset(ctx, asset)

	sdkCtx.EventManager().EmitEvent(sdk.NewEvent(
		types.EventTypeRWAMinted,
		sdk.NewAttribute(types.AttributeKeyAssetID, msg.AssetId),
		sdk.NewAttribute(types.AttributeKeyAmount, msg.Amount.String()),
		sdk.NewAttribute(types.AttributeKeyFee, feeCoins.String()),
	))
	return &types.MsgMintRWAResponse{}, nil
}

func (m msgServer) SettleRWA(ctx context.Context, msg *types.MsgSettleRWA) (*types.MsgSettleRWAResponse, error) {
	if err := msg.ValidateBasic(); err != nil {
		return nil, err
	}
	asset, found := m.GetAsset(ctx, msg.AssetId)
	if !found {
		return nil, types.ErrAssetNotFound.Wrapf("asset_id: %s", msg.AssetId)
	}
	if asset.Issuer != msg.Issuer {
		return nil, types.ErrNotIssuer.Wrapf("expected %s, got %s", asset.Issuer, msg.Issuer)
	}
	if asset.Status != types.ASSET_STATUS_ACTIVE {
		return nil, types.ErrInvalidStatus.Wrapf("must be ACTIVE to settle, got %s", asset.Status)
	}

	sdkCtx := sdk.UnwrapSDKContext(ctx)
	params, _ := m.GetParams(ctx)
	issuerAddr, _ := sdk.AccAddressFromBech32(msg.Issuer)

	// Collect settle fee
	price, _ := m.oracleKeeper.GetPrice(ctx, asset.OraclePair)
	notional := price.MulInt(asset.MintedSupply.Amount)
	feeAmt := params.SettleFeeRate.Mul(notional).TruncateInt()
	if feeAmt.IsPositive() {
		feeCoins := sdk.NewCoins(sdk.NewCoin("uvtx", feeAmt))
		if err := m.bankKeeper.SendCoinsFromAccountToModule(sdkCtx, issuerAddr, authtypes.FeeCollectorName, feeCoins); err != nil {
			return nil, err
		}
	}

	// Burn all outstanding RWA tokens
	rwaDenom := types.RWADenom(msg.AssetId)
	if asset.MintedSupply.IsPositive() {
		burnCoins := sdk.NewCoins(sdk.NewCoin(rwaDenom, asset.MintedSupply.Amount))
		// Collect from module (in practice, issuer must first send tokens back to module)
		if err := m.bankKeeper.BurnCoins(sdkCtx, types.ModuleName, burnCoins); err != nil {
			return nil, err
		}
	}

	// Release bond back to issuer
	if asset.Bond.IsPositive() {
		if err := m.bankKeeper.SendCoinsFromModuleToAccount(sdkCtx, types.ModuleName, issuerAddr, sdk.NewCoins(asset.Bond)); err != nil {
			return nil, err
		}
	}

	if err := m.SettleAsset(ctx, msg.AssetId, msg.Issuer); err != nil {
		return nil, err
	}

	sdkCtx.EventManager().EmitEvent(sdk.NewEvent(
		types.EventTypeRWASettled,
		sdk.NewAttribute(types.AttributeKeyAssetID, msg.AssetId),
		sdk.NewAttribute(types.AttributeKeyIssuer, msg.Issuer),
	))
	return &types.MsgSettleRWAResponse{}, nil
}

func (m msgServer) UpdateRestrictions(ctx context.Context, msg *types.MsgUpdateRestrictions) (*types.MsgUpdateRestrictionsResponse, error) {
	asset, found := m.GetAsset(ctx, msg.AssetId)
	if !found {
		return nil, types.ErrAssetNotFound.Wrapf("asset_id: %s", msg.AssetId)
	}
	if asset.Issuer != msg.Issuer {
		return nil, types.ErrNotIssuer.Wrapf("expected %s, got %s", asset.Issuer, msg.Issuer)
	}
	sdkCtx := sdk.UnwrapSDKContext(ctx)
	asset.Restrictions = msg.Restrictions
	asset.UpdatedAt = sdkCtx.BlockTime()
	return &types.MsgUpdateRestrictionsResponse{}, m.setAsset(ctx, asset)
}

func (m msgServer) UpdateParams(ctx context.Context, msg *types.MsgUpdateParams) (*types.MsgUpdateParamsResponse, error) {
	if msg.Authority != m.GetAuthority() {
		return nil, types.ErrNotIssuer.Wrapf("expected %s, got %s", m.GetAuthority(), msg.Authority)
	}
	return &types.MsgUpdateParamsResponse{}, m.SetParams(ctx, msg.Params)
}
